```go
// DetectFunc is the injectable function type for detecting the latest release.
type DetectFunc func(ctx context.Context, slug string) (latestVersion string, found bool, err error)

// UpdateInfo holds information about an available update.
type UpdateInfo struct {
    CurrentVersion string `json:"currentVersion"`
    LatestVersion  string `json:"latestVersion"`
    ReleaseURL     string `json:"releaseURL"`
}

// DefaultDetect uses go-selfupdate to check GitHub releases.
func DefaultDetect(ctx context.Context, slug string) (string, bool, error) {
    latest, found, err := selfupdate.DetectLatest(ctx, selfupdate.ParseSlug(slug))
    if err != nil || !found {
        return "", found, err
    }
    return latest.Version(), true, nil
}

// Check performs a rate-limited update check.
// Returns nil if no update available, version is "dev", or rate-limited.
// When force=true, bypasses the rate limit (for manual "Check for Updates").
func Check(ctx context.Context, configDir, slug, currentVersion string, detect DetectFunc, force bool) (*UpdateInfo, error)
```

Implementation details:
1. If `currentVersion == "dev" || currentVersion == ""` → return nil, nil immediately
2. If `!force`, read `configDir + "/update_check.json"`, parse `{"last_check":"RFC3339"}`, skip if within 1 hour
3. Call `detect(ctx, slug)` — if error or !found → return nil, nil
4. Compare: strip leading "v" from both versions, use `selfupdate.NewVersion(latest).GreaterThan(selfupdate.NewVersion(current))` — if not greater → return nil, nil
5. Persist `{"last_check": time.Now().UTC().Format(time.RFC3339)}` to `configDir + "/update_check.json"` (0600 perms, create dir with os.MkdirAll if needed)
6. Return `&UpdateInfo{CurrentVersion: currentVersion, LatestVersion: latestVersion, ReleaseURL: fmt.Sprintf("https://github.com/%s/releases/tag/v%s", slug, latestVersion)}`

Note: For version comparison, use `go-selfupdate`'s version comparison if available, or use `Masterminds/semver` which is likely a transitive dependency. If neither works cleanly, use a simple `selfupdate` approach: create a dummy `selfupdate.Release` and call `GreaterThan`. Check the actual API — the research shows `latest.GreaterThan(currentVersion)` which takes a string. Use that pattern.

Run tests — all must PASS.
