2. Add UpdateInfo interface (inside the file, before the component):
```tsx
interface UpdateInfo {
  currentVersion: string
  latestVersion: string
  releaseURL: string
}
```

3. Inside the WelcomeTab component function, add state and effects:
```tsx
const [update, setUpdate] = useState<UpdateInfo | null>(null)

useEffect(() => {
  // Poll bound method on mount to handle startup race
  // (event may have fired before React subscribed)
  GetLastUpdateInfo().then((info) => {
    if (info) setUpdate(info)
  }).catch(() => {}) // silent — bound method may not exist in dev

  // Subscribe to future update events from the poller
  const offUpdate = EventsOn('update:available', (info: UpdateInfo) => {
    setUpdate(info)
  })
  return () => { offUpdate() }
}, [])
```

4. Add banner JSX — render ABOVE the first section in the welcome tab content (after the logo/version area, before "Get Started" or the first `.welcome-tab__section`):
```tsx
{update && (
  <div className="update-banner" role="alert" aria-live="polite">
    <span className="update-banner__message">
      Update available: <span className="update-banner__version">{update.currentVersion}</span>
      {' '}<span className="update-banner__arrow">&rarr;</span>{' '}
      <span className="update-banner__version">{update.latestVersion}</span>
    </span>
    <div className="update-banner__actions">
      <button
        type="button"
        className="update-banner__btn--download"
        onClick={() => BrowserOpenURL(update.releaseURL)}
      >
        Download Update
      </button>
      <button
        type="button"
        className="update-banner__btn--dismiss"
        aria-label="Dismiss update notification"
        onClick={() => setUpdate(null)}
      >
        Dismiss
      </button>
    </div>
  </div>
)}
```

Place this banner inside `.welcome-tab__content`, right before the first `.welcome-tab__section` div. Check the actual JSX structure when reading the file.
