    ```typescript
    import { test, expect } from '@playwright/test'

    test.describe('Phase 93 WEB-03 + PLUG-04 push web-plugin-hot-swap', () => {
      test('initial /api/plugin-config webgl=false → no WebglAddon load on web', async ({ page, baseURL }) => {
        await page.route('**/api/plugin-config**', route => {
          // Match GET /api/plugin-config but NOT /api/plugin-config/stream (let SSE through)
          if (route.request().url().includes('/stream')) return route.continue()
          route.fulfill({
            status: 200,
            contentType: 'application/json',
            body: JSON.stringify({ webgl: false, unicode11: true, clipboard: true, search: true, webLinks: true, image: true, serialize: true, progress: false }),
          })
        })

        const url = `${baseURL}/sessions/test-session?cap=${process.env.PLAYWRIGHT_TEST_CAP}`
        await page.goto(url)
        await page.waitForTimeout(2000)
        const canvasCount = await page.locator('#terminal canvas').count()
        expect(canvasCount, 'WebGL canvas should not be present when webgl=false').toBeLessThanOrEqual(1)
      })

      test('initial /api/plugin-config webgl=true → WebglAddon loads from vendor', async ({ page, baseURL }) => {
        const webglResp = page.waitForResponse(r => r.url().includes('/assets/xterm/addons/addon-webgl.js') && r.status() === 200, { timeout: 10000 })
        const url = `${baseURL}/sessions/test-session?cap=${process.env.PLAYWRIGHT_TEST_CAP}`
        await page.goto(url)
        await webglResp
      })

      test('SSE push from /api/plugin-config/stream hot-swaps WebGL OFF without page reload (PLUG-04)', async ({ page, baseURL, request }) => {
        // 1. Initial config: webgl=true
        // 2. Page loads, WebglAddon is constructed (canvas attached)
        // 3. Test fixture flips daemon plugin settings server-side (SetPluginSettings)
        // 4. Daemon broadcasts SSE frame `event: plugin-config\ndata: {"webgl":false,...}\n\n`
        // 5. Web client receives frame, applies hot-swap (webglAddon.dispose())
        // 6. Spec asserts: NO page reload occurred AND canvas is gone within 1s of the push
        const reloadCount = { value: 0 }
        page.on('framenavigated', frame => { if (frame === page.mainFrame()) reloadCount.value++ })

        const url = `${baseURL}/sessions/test-session?cap=${process.env.PLAYWRIGHT_TEST_CAP}`
        await page.goto(url)
        await page.waitForResponse(r => r.url().includes('/assets/xterm/addons/addon-webgl.js') && r.status() === 200, { timeout: 10000 })
        const navAfterLoad = reloadCount.value
        // Confirm canvas is initially present (WebGL active)
        await page.waitForSelector('#terminal canvas', { timeout: 5000 })

        // Trigger the daemon-side flip via the test-fixture admin endpoint exposed by the Playwright webServer
        // (the fixture mirrors the daemon's SetPluginSettings RPC; the daemon then broadcasts the SSE frame)
        const flipResp = await request.post(`${baseURL}/__test__/plugin-config`, {
          data: { webgl: false, unicode11: true, clipboard: true, search: true, webLinks: true, image: true, serialize: true, progress: false },
        })
        expect(flipResp.ok(), 'test-fixture flip endpoint must succeed').toBeTruthy()

        // Wait up to 2s for the SSE-driven hot-swap to dispose the WebglAddon
        await expect(async () => {
          const cnt = await page.locator('#terminal canvas').count()
          expect(cnt, 'WebGL canvas should be removed after SSE push').toBeLessThanOrEqual(1)
        }).toPass({ timeout: 2000 })

        // Confirm: no page reload occurred during the hot-swap
        expect(reloadCount.value, 'no page reload during SSE-driven hot-swap (PLUG-04)').toBe(navAfterLoad)
      })
    })
    ```
