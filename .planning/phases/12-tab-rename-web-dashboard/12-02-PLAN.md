2. Add a `startEditById` function that enters rename mode for a specific tab ID (unlike `startEdit` which takes a Tab + event):

```tsx
function startEditById(tabId: string) {
  const tab = tabs.find(t => t.id === tabId)
  if (tab) {
    setEditingId(tab.id)
    setEditValue(tab.name)
  }
}
```

3. Add `onContextMenu` handler on the `.tab__name` span:

```tsx
onContextMenu={(e) => {
  e.preventDefault()
  e.stopPropagation()
  setContextMenu({ tabId: tab.id, x: e.clientX, y: e.clientY })
}}
```

4. Update the `.tab__name` span title from `"Double-click to rename"` to `"Double-click or right-click to rename"`.

5. Add a `useEffect` for dismiss behavior when `contextMenu` is not null:

```tsx
useEffect(() => {
  if (contextMenu === null) return
  function handleOutsideClick(e: MouseEvent) {
    setContextMenu(null)
  }
  function handleEscape(e: KeyboardEvent) {
    if (e.key === 'Escape') setContextMenu(null)
  }
  document.addEventListener('mousedown', handleOutsideClick)
  document.addEventListener('keydown', handleEscape)
  return () => {
    document.removeEventListener('mousedown', handleOutsideClick)
    document.removeEventListener('keydown', handleEscape)
  }
}, [contextMenu])
```

6. Render the floating context menu div inside the `tab-bar` div, after the `tab-bar__controls` div:

```tsx
{contextMenu && (
  <div
    className="tab__context-menu"
    role="menu"
    style={{ position: 'fixed', top: contextMenu.y, left: contextMenu.x }}
    onMouseDown={(e) => e.stopPropagation()}
  >
    <button
      role="menuitem"
      className="tab__context-menu__item"
      onClick={() => {
        startEditById(contextMenu.tabId)
        setContextMenu(null)
      }}
    >
      Rename
    </button>
  </div>
)}
```

The `onMouseDown` stopPropagation on the menu div prevents the document-level mousedown listener from closing the menu when clicking inside it.
