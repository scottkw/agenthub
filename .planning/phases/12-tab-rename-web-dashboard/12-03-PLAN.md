```javascript
function renderSessions(sessions) {
  const list = document.getElementById('session-list');
  const empty = document.getElementById('no-sessions');
  list.innerHTML = '';

  if (!sessions || sessions.length === 0) {
    empty.style.display = 'block';
    return;
  }
  empty.style.display = 'none';

  sessions.forEach(s => {
    const li = document.createElement('li');
    li.className = 'session-card';
    const id = typeof s === 'string' ? s : s.id;
    const name = (typeof s === 'object' && s.name) ? s.name : id;
    const cli = (typeof s === 'object' && s.cli_type) ? s.cli_type : '';
    const st = (typeof s === 'object' && s.status) ? s.status : 'running';
    const qrSrc = '/api/sessions/' + encodeURIComponent(id) + '/qr';
    const sessionURL = window.location.origin + '/sessions/' + encodeURIComponent(id);

    li.innerHTML = `
      <span class="session-dot session-dot--${escHtml(st)}"></span>
      <div class="session-info">
        <span class="session-name">${escHtml(name)}</span>
        <span class="session-meta">${cli ? '<span class="session-badge">' + escHtml(cli) + '</span> ' : ''}${escHtml(id)}</span>
      </div>
      <div class="session-actions">
        <img src="${qrSrc}" width="48" height="48" alt="QR code" class="qr-thumb" title="Click to enlarge QR code" data-session-id="${escHtml(id)}" data-session-url="${escHtml(sessionURL)}">
        <a href="/sessions/${escHtml(id)}" target="_blank">Open</a>
        <button class="secondary" onclick="copyTokenLink('${escHtml(id)}', this)" style="font-size:0.8rem;padding:0.35rem 0.7rem;">Copy Token Link</button>
      </div>
    `;
    list.appendChild(li);
  });

  // Attach click handlers for QR thumbnails
  list.querySelectorAll('.qr-thumb').forEach(img => {
    img.addEventListener('click', function() {
      showQROverlay(this.dataset.sessionId, this.dataset.sessionUrl);
    });
  });
}
```

Key differences from old layout:
- `li` gets `className = 'session-card'` (was bare `<li>`)
- Status dot: `session-dot session-dot--{status}` (new)
- Session info wrapper div with name and meta line (new structure)
- CLI shown as badge in meta line instead of inline span
- Session ID shown in meta line as secondary text
- Actions grouped in `session-actions` div
- QR thumb reduced from 64px to 48px for better proportion in card layout
