```go
func (ws *WebServer) handleListSessions(w http.ResponseWriter, r *http.Request) {
    ids := ws.webEnabledSessions()
    items := make([]sessionListItem, 0, len(ids))
    for _, id := range ids {
        name, cliType, st := "", "", ""
        if ws.sessionResolver != nil {
            name, cliType, st = ws.sessionResolver(id)
        }
        if name == "" {
            name = id
        }
        items = append(items, sessionListItem{ID: id, Name: name, CLIType: cliType, Status: st})
    }
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(items) //nolint:errcheck
}
```

In `app.go`, in the `StartWebServer` method, after the line `ws.LoadPasswordHash(hash)` (around line 389) and before `ws.Start()`, add:

```go
ws.SetSessionResolver(func(id string) (string, string, string) {
    a.mu.RLock()
    defer a.mu.RUnlock()
    name := a.tabNames[id]
    cliType := ""
    // Resolve CLI type from session registry
    if sess, ok := a.registry.Get(id); ok {
        cliType = sess.CLI
    }
    st := ""
    if s, ok := a.sessionStatuses[id]; ok {
        st = string(s)
    }
    return name, cliType, st
})
```

This closure captures `a` and reads from `a.tabNames`, `a.registry`, and `a.sessionStatuses` at query time (not definition time), protected by `a.mu.RLock()`. The registry's own Get method is thread-safe, but CLI is read after Get returns the Session struct.

Note: Verify that `a.registry` is the field name for the SessionRegistry. If it's named differently (e.g., `a.sessions` or `a.backend`), adapt accordingly. Check the App struct definition at the top of app.go.
