```go
//go:build e2e

package webserver

import (
    "context"
    "fmt"
    "os"
    "path/filepath"
    "testing"
    "time"

    "github.com/chromedp/chromedp"
)

// TestFindBar_10kPerf asserts that addon-search.findNext completes within
// 1000ms wall-clock against a 10,000-line scrollback. Encodes ROADMAP
// Phase 94 SC-3: "Search across a 10,000-line scrollback fixture completes
// without UI lockup (no >1s frame budget breach)."
//
// Mitigates T-94-04: regex catastrophic backtracking. Test uses plain-string
// search (not pathological regex); pathological regex is documented as
// accepted limitation per RESEARCH Pitfall #5.
func TestFindBar_10kPerf(t *testing.T) {
    // Step 1: load fixture.
    fixturePath := filepath.Join("testdata", "findbar_perf_fixture.txt")
    fixtureBytes, err := os.ReadFile(fixturePath)
    if err != nil {
        t.Fatalf("read fixture: %v", err)
    }
    if lineCount := /* count lines */; lineCount != 10000 {
        t.Fatalf("fixture has %d lines, want 10000 (T-94-04 perf budget invariant)", lineCount)
    }

    // Step 2: spin up the test webserver (uses existing testServer helper).
    ws, _ := testServer(t)

    // Step 3: chromedp headless tab.
    ctx, cancel := chromedp.NewContext(context.Background())
    defer cancel()
    ctx, cancel = context.WithTimeout(ctx, 30*time.Second)
    defer cancel()

    // Step 4: navigate to a test harness HTML that loads xterm + addon-search
    // the same way terminal.html does, exposes term + searchAddon as window.*
    // for the test, and skips the WebSocket/PTY layer.
    // Reuse existing terminal.html if it exposes window.term + window.searchAddon
    // OR add a dedicated /test-harness route in the test webserver.
    // For simplicity, navigate to terminal.html with a no-op session id and
    // poll for window.term to exist; then call writeSync + findNext via JS eval.
    target := ws.BaseURL() + "/sessions/test-perf-fixture"  // adapt per existing route shape

    var elapsedMs float64
    err = chromedp.Run(ctx,
        chromedp.Navigate(target),
        // Wait for term to be ready
        chromedp.Poll("typeof window.term !== 'undefined' && typeof window.searchAddon !== 'undefined'", nil, chromedp.WithPollingTimeout(15*time.Second)),
        // Write the 10k-line fixture into the xterm buffer via the terminal API.
        chromedp.Evaluate(fmt.Sprintf(`
            (function() {
                window.term.write(%q);
            })();
        `, string(fixtureBytes)), nil),
        // Wait for write to complete (term.write is async — poll for buffer length).
        chromedp.Poll(`window.term.buffer.active.length >= 10000`, nil, chromedp.WithPollingTimeout(10*time.Second)),
        // Run findNext under performance.now() and capture duration.
        chromedp.Evaluate(`
            (function() {
                const t0 = performance.now()
                window.searchAddon.findNext('needle', { regex: false, caseSensitive: false, wholeWord: false })
                return performance.now() - t0
            })()
        `, &elapsedMs),
    )
    if err != nil {
        t.Fatalf("chromedp run: %v", err)
    }

    if elapsedMs > 1000.0 {
        t.Fatalf("findNext over 10k-line scrollback took %.2fms; budget is 1000ms (T-94-04 mitigation; ROADMAP SC-3)", elapsedMs)
    }
    t.Logf("findNext over 10k-line scrollback: %.2fms (budget 1000ms)", elapsedMs)
}
```

import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest'
import React from 'react'
import { createRoot, type Root } from 'react-dom/client'
import { flushSync } from 'react-dom'
import { FindBar } from '../FindBar'
import terminalPanelSrc from '../../TerminalPanel.tsx?raw'

// Phase 94 SRC-03 cancellation contract — Pitfall #10 mitigation.
describe('FindBar — cancellation (SRC-03)', () => {
  let container: HTMLElement
  let root: Root | null

  beforeEach(() => {
    container = document.createElement('div')
    document.body.appendChild(container)
    root = null
    vi.useFakeTimers()
  })

  afterEach(() => {
    if (root) flushSync(() => root!.unmount())
    container.remove()
    vi.useRealTimers()
  })

  it('clicking close button invokes onClose synchronously', () => {
    const onClose = vi.fn()
    root = createRoot(container)
    flushSync(() => {
      root!.render(
        React.createElement(FindBar, {
          query: 'foo',
          onQueryChange: vi.fn(),
          matchCount: 5,
          currentMatchIndex: 2,
          searchOptions: { regex: false, caseSensitive: false, wholeWord: false },
          onSearchOptionsChange: vi.fn(),
          onNext: vi.fn(),
          onPrev: vi.fn(),
          onClose,
        })
      )
    })
    const closeBtn = container.querySelector<HTMLButtonElement>('.find-bar__close')
    expect(closeBtn).not.toBeNull()
    closeBtn!.click()
    expect(onClose).toHaveBeenCalledTimes(1)
  })

  it('Esc on the find-bar container invokes onClose (Pitfall #3 — handler at container level)', () => {
    const onClose = vi.fn()
    root = createRoot(container)
    flushSync(() => {
      root!.render(
        React.createElement(FindBar, {
          query: '',
          onQueryChange: vi.fn(),
          matchCount: 0,
          currentMatchIndex: -1,
          searchOptions: { regex: false, caseSensitive: false, wholeWord: false },
          onSearchOptionsChange: vi.fn(),
          onNext: vi.fn(),
          onPrev: vi.fn(),
          onClose,
        })
      )
    })
    // Focus a non-input element inside the find bar (a toggle), simulating
    // Pitfall #3: Esc must work even when input is not focused.
    const caseToggle = container.querySelector<HTMLButtonElement>('.find-bar__toggle--case')
    expect(caseToggle).not.toBeNull()
    caseToggle!.focus()
    const escEvent = new KeyboardEvent('keydown', { key: 'Escape', bubbles: true })
    caseToggle!.dispatchEvent(escEvent)
    expect(onClose).toHaveBeenCalledTimes(1)
  })

  })
})
