    ```
    package tui

    import (
        "context"
        "time"

        "github.com/scottkw/agenthub/internal/daemon"
        // import "github.com/scottkw/agenthub/internal/files" if the existing TUI code uses files.FileEntry
    )

    // FilesClient is the narrow interface used by the TUI Files view. Both the
    // local *daemon.DaemonClient (Unix socket / TCP relay) and the Phase 122
    // *RemoteFilesClient (HTTPS + cap via the local daemon proxy) satisfy it.
    type FilesClient interface {
        ListFiles(ctx context.Context, sessionID, relPath string) (entries []<FileEntryType>, truncated bool, err error)
        StatFile(ctx context.Context, sessionID, relPath string) (<FileEntryType>, error)
        ReadFile(ctx context.Context, sessionID, relPath string) (data []byte, mime string, err error)
        HeadFile(ctx context.Context, sessionID, relPath string) (size int64, mime string, mtime time.Time, err error)
    }

    // Compile-time assertion. If *daemon.DaemonClient's signatures drift, this
    // line fails to build — keeping the contract honest.
    var _ FilesClient = (*daemon.DaemonClient)(nil)
    ```
