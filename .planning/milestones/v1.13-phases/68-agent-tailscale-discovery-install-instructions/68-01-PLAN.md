```go
//go:build windows

package daemon

import (
	"os"
	"path/filepath"
)

// platformExtraBins returns Windows-specific binary directories for agent CLIs
// and Tailscale. Called from AugmentServicePath in path.go.
func platformExtraBins() []string {
	var paths []string
	if appdata := os.Getenv("APPDATA"); appdata != "" {
		paths = append(paths, filepath.Join(appdata, "npm"))
	}
	if local := os.Getenv("LOCALAPPDATA"); local != "" {
		paths = append(paths, filepath.Join(local, "pnpm"))
		paths = append(paths, filepath.Join(local, "Programs", "nodejs"))
	}
	paths = append(paths, `C:\Program Files\Tailscale`)
	return paths
}
```

```go
func TestAugmentServicePath_Cargo(t *testing.T) {
	if runtime.GOOS == "windows" {
		t.Skip("uses Unix PATH separator")
	}
	home := t.TempDir()
	t.Setenv("HOME", home)
	original := "/usr/bin:/bin"
	t.Setenv("PATH", original)

	cargoBin := filepath.Join(home, ".cargo", "bin")
	if err := os.MkdirAll(cargoBin, 0755); err != nil {
		t.Fatalf("MkdirAll: %v", err)
	}

	AugmentServicePath()

	got := os.Getenv("PATH")
	if !strings.Contains(got, cargoBin) {
		t.Errorf("PATH should contain %s, got %s", cargoBin, got)
	}
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| PATH environment variable | Directories added to PATH affect which binaries are executed by the daemon |
| Filesystem stat | os.Stat checks whether directories exist before adding to PATH |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-68-01 | Tampering | PATH augmentation | accept | Directories added are well-known system/user paths with standard permissions. The daemon already runs as the current user; adding `~/.cargo/bin` or `/snap/bin` to PATH does not elevate privileges beyond what the user already has. No user-controlled input determines which paths are added (hardcoded list). |
| T-68-02 | Elevation of Privilege | platformExtraBins `C:\Program Files\Tailscale` | accept | This is a standard system-protected directory. Only administrators can write to it. Adding it to PATH for binary lookup is standard practice. |
| T-68-03 | Information Disclosure | PATH contents visible in env | accept | PATH is not sensitive. It contains directory paths, not credentials. Logging PATH for debugging is acceptable. |
</threat_model>
