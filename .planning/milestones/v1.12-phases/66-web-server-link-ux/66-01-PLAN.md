```typescript
async function handleCopyURL() {
  if (!serverURL) return
  await ClipboardSetText(serverURL)
  setUrlCopied(true)
  setTimeout(() => setUrlCopied(false), 1500)
}

async function handleToggleDashQR() {
  if (showDashQR) {
    setShowDashQR(false)
    return
  }
  setQrError(null)
  if (!dashQRb64) {
    try {
      const b64 = await GetWebServerQRCode()
      setDashQRb64(b64)
    } catch {
      setQrError('QR unavailable \u2014 tap to retry')
      return
    }
  }
  setShowDashQR(true)
}
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-66-01 | Information Disclosure | QR code / copy exposes dashboard URL | accept | URL is already displayed as plaintext in Settings. QR/copy are convenience affordances for an already-visible value. No new information exposed. |
| T-66-02 | Spoofing | Clipboard write could be intercepted | accept | `ClipboardSetText` routes through native Wails runtime (not browser API). Clipboard interception is an OS-level concern, not in app's control. |
| T-66-03 | Denial of Service | Repeated QR generation calls | accept | QR is cached in React state after first fetch. Even uncached, `qrcode.Encode` for a short URL is sub-millisecond on any hardware. No amplification vector. |
</threat_model>
