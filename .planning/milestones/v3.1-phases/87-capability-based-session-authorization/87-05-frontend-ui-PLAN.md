       ```tsx
       import React, { useState } from 'react'
       import { GetCapabilityQRCode } from '../wailsjs/go/main/App'
       import { BrowserOpenURL, ClipboardSetText } from '../wailsjs/wailsjs/runtime/runtime'

       interface SessionSharePanelProps {
         sessionId: string
         readURL: string
         writeURL: string
         readCode: string
         writeCode: string
       }

       export function SessionSharePanel({ sessionId, readURL, writeURL, readCode, writeCode }: SessionSharePanelProps) {
         const [readCopied, setReadCopied] = useState(false)
         const [writeCopied, setWriteCopied] = useState(false)
         const [showReadQR, setShowReadQR] = useState(false)
         const [showWriteQR, setShowWriteQR] = useState(false)
         const [readQRb64, setReadQRb64] = useState<string | null>(null)
         const [writeQRb64, setWriteQRb64] = useState<string | null>(null)
         const [qrError, setQrError] = useState<string | null>(null)

         async function handleCopy(url: string, setter: (v: boolean) => void) {
           await ClipboardSetText(url)
           setter(true)
           setTimeout(() => setter(false), 1500)
         }

         async function handleToggleQR(which: 'read' | 'write') {
           // Close the other QR (only one visible at a time per UI-SPEC Surface 1).
           if (which === 'read') {
             if (showReadQR) { setShowReadQR(false); return }
             setShowWriteQR(false)
             setQrError(null)
             if (!readQRb64) {
               try {
                 // joinExchangeURL origin derived from readURL host, path /join?code=<readCode>
                 const u = new URL(readURL)
                 const joinURL = `${u.protocol}//${u.host}/join?code=${readCode}`
                 const b64 = await GetCapabilityQRCode(joinURL)
                 setReadQRb64(b64)
               } catch { setQrError('QR unavailable — tap to retry'); return }
             }
             setShowReadQR(true)
           } else {
             if (showWriteQR) { setShowWriteQR(false); return }
             setShowReadQR(false)
             setQrError(null)
             if (!writeQRb64) {
               try {
                 const u = new URL(writeURL)
                 const joinURL = `${u.protocol}//${u.host}/join?code=${writeCode}`
                 const b64 = await GetCapabilityQRCode(joinURL)
                 setWriteQRb64(b64)
               } catch { setQrError('QR unavailable — tap to retry'); return }
             }
             setShowWriteQR(true)
           }
         }

         return (
           <div className="session-share-panel" data-session-id={sessionId}>
             <div className="session-share-panel__link-row">
               <span className="session-share-panel__label">Read-Only Link</span>
               <span className="session-share-panel__url" title={readURL}>{readURL}</span>
               <div className="session-share-panel__actions">
                 <button className="daemon-panel__btn" onClick={() => handleCopy(readURL, setReadCopied)}>{readCopied ? 'Copied!' : 'Copy'}</button>
                 <button className="daemon-panel__btn" onClick={() => BrowserOpenURL(readURL)}>Open</button>
                 <button className="daemon-panel__btn" onClick={() => handleToggleQR('read')}>{showReadQR ? 'Hide QR' : 'QR'}</button>
               </div>
             </div>
             {showReadQR && readQRb64 && (
               <img className="session-share-panel__qr" src={`data:image/png;base64,${readQRb64}`} width={200} height={200} alt="QR code for read-only share link" />
             )}

             <div className="session-share-panel__link-row">
               <span className="session-share-panel__label">Full Access Link</span>
               <span className="session-share-panel__url" title={writeURL}>{writeURL}</span>
               <div className="session-share-panel__actions">
                 <button className="daemon-panel__btn" onClick={() => handleCopy(writeURL, setWriteCopied)}>{writeCopied ? 'Copied!' : 'Copy'}</button>
                 <button className="daemon-panel__btn" onClick={() => BrowserOpenURL(writeURL)}>Open</button>
                 <button className="daemon-panel__btn" onClick={() => handleToggleQR('write')}>{showWriteQR ? 'Hide QR' : 'QR'}</button>
               </div>
             </div>
             {showWriteQR && writeQRb64 && (
               <img className="session-share-panel__qr" src={`data:image/png;base64,${writeQRb64}`} width={200} height={200} alt="QR code for full-access share link" />
             )}

             {qrError && <p className="session-share-panel__error">{qrError}</p>}
           </div>
         )
       }
       ```

       ```tsx
       import React, { useEffect, useRef, useState } from 'react'

       interface RegenerateKeyModalProps {
         isOpen: boolean
         onConfirm: () => Promise<void>
         onCancel: () => void
       }

       export function RegenerateKeyModal({ isOpen, onConfirm, onCancel }: RegenerateKeyModalProps) {
         const [acting, setActing] = useState(false)
         const [error, setError] = useState<string | null>(null)
         const cancelBtnRef = useRef<HTMLButtonElement>(null)

         useEffect(() => {
           if (!isOpen) return
           function handleKeyDown(e: KeyboardEvent) { if (e.key === 'Escape') onCancel() }
           window.addEventListener('keydown', handleKeyDown)
           return () => window.removeEventListener('keydown', handleKeyDown)
         }, [isOpen, onCancel])

         useEffect(() => { if (isOpen) cancelBtnRef.current?.focus() }, [isOpen])

         useEffect(() => { if (!isOpen) { setError(null); setActing(false) } }, [isOpen])

         if (!isOpen) return null

         async function handleConfirm() {
           setError(null)
           setActing(true)
           try {
             await onConfirm()
           } catch (err) {
             setError(err instanceof Error ? err.message : String(err))
             setActing(false)
             return
           }
           setActing(false)
         }

         return (
           <div className="quit-modal-overlay" role="dialog" aria-modal="true" aria-labelledby="regen-key-title">
             <div className="quit-modal">
               <div className="quit-modal__header">
                 <h2 id="regen-key-title">Regenerate Signing Key?</h2>
                 <button className="quit-modal__close" onClick={onCancel} aria-label="Close">×</button>
               </div>
               <div className="quit-modal__body">
                 <p>This immediately invalidates ALL shared links across ALL sessions. Anyone currently using a shared terminal will be disconnected. You will need to re-share sessions to give access again.</p>
                 {error && <p className="quit-modal__error">{error}</p>}
               </div>
               <div className="quit-modal__footer">
                 <button ref={cancelBtnRef} className="quit-modal__btn--cancel" onClick={onCancel} disabled={acting}>Keep Links</button>
                 <button className="quit-modal__btn--quit-all" onClick={handleConfirm} disabled={acting}>{acting ? 'Invalidating…' : 'Invalidate All Links'}</button>
               </div>
             </div>
           </div>
         )
       }
       ```
